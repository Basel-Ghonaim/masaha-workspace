import type { ReactNode } from 'react';
import { useLocation, useRevalidator, useRouteError } from 'react-router';
import { classifyRouteError } from './classifyRouteError';
import { reloadPage } from './reloadPage';
import { RetryState } from './RetryState';

// The page a retry of the general error ran again (its location's key): the boundary is mounted
// anew when the error comes back, so what it remembers lives outside it.
let retried: string | null = null;

/**
 * A route's error boundary: offline when the page's code could not arrive or the connection is
 * gone, the general error otherwise. Offline, "Try again" reloads the page (a failed lazy load is
 * kept by the router, so only a fresh page fetches the code again), and so does the connection
 * coming back; for the general error it runs the route again in place, and a second failure in a
 * row says «ما زال الخطأ قائمًا. حاول بعد قليل.» (G8.1). `children` are the shell's other ways on,
 * after "Try again".
 */
export function RouteErrorState({ children }: { children?: ReactNode }) {
  const kind = classifyRouteError(useRouteError(), navigator.onLine);
  const { key } = useLocation();
  const revalidator = useRevalidator();
  const again = kind === 'error' && retried === key;

  return (
    <RetryState
      kind={kind}
      again={again}
      onRetry={
        kind === 'offline'
          ? reloadPage
          : () => {
              retried = key;
              void revalidator.revalidate();
            }
      }
    >
      {children}
    </RetryState>
  );
}
